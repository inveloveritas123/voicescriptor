var currentResult = null;
var logEntries = [];
var chVideos = [];

// ═══ Helpers ═══
function esc(s){var d=document.createElement("div");d.textContent=s;return d.innerHTML}
function toast(m){var t=document.getElementById("toast");t.textContent=m;t.classList.add("show");setTimeout(function(){t.classList.remove("show")},3000)}
function addLog(m){var ts=new Date().toLocaleTimeString("de-DE",{hour:"2-digit",minute:"2-digit",second:"2-digit"});logEntries.push("["+ts+"] "+m);if(logEntries.length>100)logEntries=logEntries.slice(-100);renderLog()}
function renderLog(){var el=document.getElementById("logList");if(!el)return;el.innerHTML=logEntries.slice().reverse().map(function(l){var c=l.indexOf("FEHLER")>=0?"var(--err)":l.indexOf("OK")>=0?"var(--green)":"var(--sec)";return"<div style='padding:1px 0;color:"+c+"'>"+esc(l)+"</div>"}).join("")}
function togglePanel(id){var el=document.getElementById(id);if(!el)return;el.classList.toggle("hidden");if(id==="history"&&!el.classList.contains("hidden"))loadHistory()}
function showProg(t,p){document.getElementById("progress").classList.remove("hidden");document.getElementById("progText").textContent=t;document.getElementById("progBar").style.width=p+"%"}
function hideProg(){document.getElementById("progress").classList.add("hidden")}

// ═══ Transcribe ═══
async function doTranscribe(){
  var url=document.getElementById("url").value.trim();
  if(!url){toast("URL eingeben");return}
  var mode=document.querySelector("input[name=mode]:checked").value;
  var sum=document.getElementById("autoSum").checked;
  showProg("Verbinde...",5);
  document.getElementById("result").classList.add("hidden");
  addLog("Transkription: "+url+" ("+mode+")");
  var poll=setInterval(function(){fetch("/api/progress/current").then(function(r){return r.json()}).then(function(d){if(d&&d.step!=="idle")showProg(d.detail||d.step,d.pct||0)}).catch(function(){})},2000);
  try{
    var r=await fetch("/api/transcribe",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({url:url,mode:mode,model_size:"base",language:"",auto_summary:sum})});
    clearInterval(poll);
    if(!r.ok){var e=await r.json().catch(function(){return{}});showProg("Fehler",100);addLog("FEHLER: "+(e.detail||"HTTP "+r.status));toast("Fehler: "+(e.detail||r.status));return}
    showProg("Fertig!",100);
    var d=await r.json();
    currentResult=d;
    addLog("OK: "+(d.method||d.title||""));
    document.getElementById("resTitle").textContent=d.title||"Ergebnis";
    document.getElementById("resMeta").textContent=(d.method||"")+" | "+(d.channel||"")+" | "+Math.round((d.transcript.duration||0)/60)+" Min";
    document.getElementById("resText").textContent=d.transcript.full_text||"";
    document.getElementById("result").classList.remove("hidden");
    toast("Transkription fertig");
    setTimeout(hideProg,2000);
  }catch(e){clearInterval(poll);showProg("Fehler",100);addLog("FEHLER: "+e.message);toast(e.message)}
}
function showTab(t){if(!currentResult)return;var el=document.getElementById("resText");if(t==="srt"){var s=currentResult.transcript.segments||[];el.textContent=s.map(function(x,i){return(i+1)+"\n"+(x.start_formatted||"00:00:00")+" --> "+(x.end_formatted||"00:00:00")+"\n"+x.text}).join("\n\n")}else{el.textContent=currentResult.transcript.full_text||""}}
function copyResult(){if(currentResult)navigator.clipboard.writeText(currentResult.transcript.full_text);toast("Kopiert")}
async function doInfo(){
  var url=document.getElementById("url").value.trim();
  if(!url)return;
  try{var r=await fetch("/api/info",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({url:url})});var d=await r.json();toast(d.title+" | "+d.channel+" | "+Math.round((d.duration||0)/60)+"min")}catch(e){toast("Info-Fehler")}
}

// ═══ History ═══
async function loadHistory(){
  try{
    var r=await fetch("/api/transcripts");var items=await r.json();
    var el=document.getElementById("histList");
    if(!items.length){el.innerHTML="<div style='color:var(--muted)'>Noch keine Transkripte</div>";return}
    el.innerHTML=items.slice(0,10).map(function(t){
      var dur=t.duration?Math.round(t.duration/60)+" Min":"";
      var date=t.created_at?new Date(t.created_at).toLocaleDateString("de-DE")+" "+new Date(t.created_at).toLocaleTimeString("de-DE",{hour:"2-digit",minute:"2-digit"}):"";
      var ch=t.channel?"<span style='color:var(--accent)'>"+esc(t.channel)+"</span> · ":"";
      var media=(t.media_files||[]).map(function(f){return"<a href='"+f.url+"' download class='btn btn-sm btn-outline' style='font-size:10px'>"+f.name.split(".").pop().toUpperCase()+"</a>"}).join(" ");
      return"<div class='hist-item'><div style='display:flex;justify-content:space-between'><div><b>"+esc(t.title||"?")+"</b><br><span style='font-size:10px;color:var(--sec)'>"+ch+(t.method||t.mode||"")+" · "+dur+" · "+date+"</span></div><button class='btn btn-err btn-sm' onclick=\"delTranscript('"+t.job_id+"')\">&#10005;</button></div><div style='display:flex;gap:4px;margin-top:4px;flex-wrap:wrap'><a href='/api/transcripts/"+encodeURIComponent(t.filename)+"?format=txt' download class='btn btn-sm btn-outline' style='font-size:10px'>TXT</a><a href='/api/transcripts/"+encodeURIComponent(t.filename)+"?format=srt' download class='btn btn-sm btn-outline' style='font-size:10px'>SRT</a>"+media+"</div></div>"
    }).join("");
  }catch(e){console.error(e)}
}
async function delTranscript(id){if(!confirm("Loeschen?"))return;await fetch("/api/transcripts/"+id,{method:"DELETE"});loadHistory()}

// ═══ Channel Videos ═══
async function loadChannel(){
  var url=document.getElementById("chUrl").value.trim();if(!url)return;
  var el=document.getElementById("chList");el.innerHTML="<span style='color:var(--accent)'>Lade...</span>";
  addLog("Channel laden: "+url);
  try{
    var r=await fetch("/api/channel/videos",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({url:url,limit:10})});
    var d=await r.json();
    if(d.error){el.innerHTML="<span style='color:var(--err)'>"+esc(d.error)+"</span>";addLog("FEHLER: "+d.error);return}
    chVideos=d.videos||[];
    if(!chVideos.length){el.innerHTML="Keine Videos";return}
    el.innerHTML="<div style='font-size:11px;color:var(--sec);margin-bottom:4px'>"+esc(d.channel||"Channel")+" — "+chVideos.length+" Videos</div>"+chVideos.map(function(v,i){
      var dur=v.duration?Math.floor(v.duration/60)+":"+String(Math.floor(v.duration%60)).padStart(2,"0"):"?";
      return"<label style='display:flex;gap:6px;padding:4px 0;border-bottom:1px solid var(--border);cursor:pointer;align-items:flex-start'><input type='checkbox' class='vck' data-i='"+i+"' checked style='margin-top:3px'><div style='flex:1'><div style='font-size:12px'>"+esc(v.title)+"</div><div style='font-size:10px;color:var(--muted)'>"+dur+"</div></div><a href='"+v.url+"' target='_blank' style='font-size:10px'>&#8599;</a></label>"
    }).join("");
    document.getElementById("batchCtrl").classList.remove("hidden");
    addLog("Channel: "+chVideos.length+" Videos geladen");
  }catch(e){el.innerHTML="<span style='color:var(--err)'>"+e.message+"</span>"}
}
function selAll(v){document.querySelectorAll(".vck").forEach(function(c){c.checked=v})}
async function batchTranscribe(){
  var sel=[];document.querySelectorAll(".vck:checked").forEach(function(c){sel.push(chVideos[parseInt(c.dataset.i)])});
  if(!sel.length){toast("Keine ausgewaehlt");return}
  var st=document.getElementById("batchSt");
  for(var i=0;i<sel.length;i++){
    st.textContent=(i+1)+"/"+sel.length+": "+sel[i].title.slice(0,35)+"...";
    showProg("Batch "+(i+1)+"/"+sel.length,Math.round(i/sel.length*100));
    addLog("Batch "+(i+1)+"/"+sel.length+": "+sel[i].title.slice(0,40));
    try{await fetch("/api/transcribe",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({url:sel[i].url,mode:"auto",auto_summary:false})})}catch(e){addLog("FEHLER: "+sel[i].title.slice(0,30))}
  }
  st.textContent="Fertig! "+sel.length+" transkribiert";showProg("Batch fertig",100);addLog("Batch fertig: "+sel.length+" Videos");setTimeout(hideProg,3000)
}

// ═══ Cookies ═══
async function refreshCookies(){
  try{
    var r=await fetch("/api/cookies/status");var d=await r.json();
    var b=document.getElementById("cookieBadge");var s=document.getElementById("cookieStatus");
    if(d.exists){b.textContent="AKTIV";b.style.background="rgba(34,197,94,.15)";b.style.color="var(--green)";s.textContent=d.entries+" Cookies aktiv"+(d.modified?" ("+d.modified+")":"")}
    else{b.textContent="FEHLT";b.style.background="rgba(245,158,11,.15)";b.style.color="var(--warn)";s.textContent="Keine Cookies"}
    fetch("/api/cookies/history").then(function(r){return r.json()}).then(function(h){if(h.length){var l=h[h.length-1];document.getElementById("cookieHist").textContent="Letzter Import: "+l.timestamp+" ("+l.method+")"}}).catch(function(){})
  }catch(e){}
}
async function doHarvest(){
  document.getElementById("cookieBadge").textContent="...";addLog("Auto-Harvest...");toast("Harvest laeuft (15-30s)");
  try{var r=await fetch("/api/cookies/harvest",{method:"POST"});var d=await r.json();if(r.ok){addLog("Harvest OK: "+d.cookie_count+" Cookies");toast("Cookies: "+d.cookie_count)}else{addLog("FEHLER: "+(d.detail||"?"));toast("Fehler: "+(d.detail||"?"))}}catch(e){addLog("FEHLER: "+e.message);toast(e.message)}
  refreshCookies()
}
async function uploadCookieFile(){
  var f=document.getElementById("cookieFile");if(!f.files.length){toast("Keine Datei");return}
  var fd=new FormData();fd.append("file",f.files[0]);
  try{var r=await fetch("/api/cookies/upload",{method:"POST",body:fd});var d=await r.json();if(d.ok||d.status==="ok"){toast("Cookies geladen: "+(d.entries||d.cookie_count))}else toast("Fehler")}catch(e){toast(e.message)}
  refreshCookies()
}
async function saveCookiePaste(){
  var t=document.getElementById("cookiePaste").value.trim();if(!t){toast("Kein Text");return}
  var resp;
  if(t.indexOf("\t")>=0||t.indexOf("# ")===0){var b=new Blob([t],{type:"text/plain"});var fd=new FormData();fd.append("file",b,"cookies.txt");resp=await fetch("/api/cookies/upload",{method:"POST",body:fd})}
  else{resp=await fetch("/api/cookies/from-browser",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({cookies:t})})}
  var d=await resp.json();if(d.ok){toast("Gespeichert! "+(d.entries||"")+" Eintraege");document.getElementById("cookiePaste").value="";document.getElementById("cookiePaste").style.borderColor="var(--green)";setTimeout(function(){document.getElementById("cookiePaste").style.borderColor=""},2000)}else toast("Fehler");
  refreshCookies()
}
async function doCookieDelete(){if(!confirm("Cookies loeschen?"))return;await fetch("/api/cookies",{method:"DELETE"});toast("Geloescht");refreshCookies()}
async function doCookieLogout(){if(!confirm("Logout?"))return;await fetch("/api/cookies",{method:"DELETE"});toast("Ausgeloggt");refreshCookies()}

// ═══ Google Login + 2FA ═══
async function doGoogleLogin(){
  var email=document.getElementById("gEmail").value.trim();var pw=document.getElementById("gPass").value;
  if(!email||!pw){toast("Email+Passwort eingeben");return}
  var btn=document.getElementById("gLoginBtn");btn.textContent="Logge ein...";btn.disabled=true;addLog("Google Login: "+email);
  try{
    var r=await fetch("/api/cookies/google-login",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({email:email,password:pw})});
    var d=await r.json();
    if(d.status==="ok"){addLog("Google OK: "+d.cookie_count+" Cookies");toast("Eingeloggt!");togglePanel("glogin");refreshCookies();document.getElementById("gPass").value=""}
    else if(d.status==="2fa_required"){addLog("2FA: "+d.message);show2fa(d.screenshot,d.message)}
    else{addLog("FEHLER: "+(d.detail||d.message||"?"));toast("Fehler: "+(d.detail||d.message||"?"))}
  }catch(e){addLog("FEHLER: "+e.message);toast(e.message)}
  btn.textContent="Einloggen";btn.disabled=false
}
function show2fa(img,hint){
  document.getElementById("tfaBox").classList.remove("hidden");
  if(img){document.getElementById("tfaImg").src="data:image/png;base64,"+img;document.getElementById("tfaImg").style.display="block"}
  document.getElementById("tfaHint").textContent=hint||"2FA Code eingeben"
}
async function submit2fa(){
  var code=document.getElementById("tfaCode").value.trim();if(!code)return;addLog("2FA Code...");
  try{var r=await fetch("/api/cookies/google-2fa",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({code:code,action:"code"})});var d=await r.json();
  if(d.status==="ok"){addLog("2FA OK!");toast("Eingeloggt!");document.getElementById("tfaBox").classList.add("hidden");togglePanel("glogin");refreshCookies()}
  else if(d.status==="2fa_required"){show2fa(d.screenshot,d.message)}
  else{addLog("FEHLER: "+(d.message||"?"));toast(d.message||"?")}}catch(e){toast(e.message)}
}
